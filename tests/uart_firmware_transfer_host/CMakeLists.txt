cmake_minimum_required(VERSION 3.22)
project(uart_firmware_transfer_host LANGUAGES C CXX)
enable_testing()
include(../../firmware_common/pigweed_transfer_host.cmake)
add_library(firmware_client STATIC ../../tools/native/firmware_transfer_client.cc
    ../../firmware_common/uart_firmware_transfer_hex.c)
target_include_directories(firmware_client PUBLIC ../../tools/native ../../firmware_common)
target_link_libraries(firmware_client PUBLIC pigweed_transfer_host)
target_compile_options(firmware_client PRIVATE -Wall -Wextra -Werror)
add_executable(firmware_transfer_client ../../tools/native/firmware_transfer_stdio.cc)
target_link_libraries(firmware_transfer_client PRIVATE firmware_client)
if(WIN32)
    target_link_options(firmware_transfer_client PRIVATE -static)
endif()
add_executable(firmware_transfer_test test.cpp ../../firmware_common/uart_firmware_transfer.cc)
target_link_libraries(firmware_transfer_test PRIVATE firmware_client)
target_compile_definitions(firmware_transfer_test PRIVATE PW_TRANSFER_HOST=1)
target_compile_options(firmware_transfer_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
foreach(case normal tail full drop_data drop_reply malformed wrong_crc write_failure close_active local_only not_prepared)
    add_test(NAME uart_firmware_${case} COMMAND firmware_transfer_test ${case})
    set_tests_properties(uart_firmware_${case} PROPERTIES TIMEOUT 50)
endforeach()
add_executable(firmware_transfer_stdio_server stdio_server.cpp
    ../../firmware_common/uart_firmware_transfer.cc)
target_link_libraries(firmware_transfer_stdio_server PRIVATE firmware_client)
target_compile_definitions(firmware_transfer_stdio_server PRIVATE PW_TRANSFER_HOST=1)
target_compile_options(firmware_transfer_stdio_server PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(WIN32)
    target_link_options(firmware_transfer_stdio_server PRIVATE -static)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME uart_firmware_stdio_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/stdio_integration.py
    $<TARGET_FILE:firmware_transfer_stdio_server> $<TARGET_FILE:firmware_transfer_client>
    ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(uart_firmware_stdio_integration PROPERTIES TIMEOUT 40)
