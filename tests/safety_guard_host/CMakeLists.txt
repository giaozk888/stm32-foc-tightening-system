cmake_minimum_required(VERSION 3.22)
project(safety_guard_host LANGUAGES C)
enable_testing()
add_executable(safety_guard_test contract_test.c ../../firmware_common/safety_guard.c)
target_include_directories(safety_guard_test PRIVATE ../../firmware_common)
target_compile_features(safety_guard_test PRIVATE c_std_11)
if(MSVC)
    target_compile_options(safety_guard_test PRIVATE /W4 /WX)
else()
    target_compile_options(safety_guard_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME safety_guard_test COMMAND safety_guard_test)
