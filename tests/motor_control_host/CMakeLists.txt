cmake_minimum_required(VERSION 3.22)
project(motor_control_host LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
add_library(motor_control STATIC ../../firmware_common/motor_control.c)
target_include_directories(motor_control PUBLIC ../../firmware_common)
target_compile_options(motor_control PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
add_executable(engine_test engine_test.cpp)
target_link_libraries(engine_test PRIVATE motor_control)
target_compile_options(engine_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
add_test(NAME motor_control_admission_replay_and_timeout COMMAND engine_test)
add_executable(port_test port_test.c)
target_link_libraries(port_test PRIVATE motor_control)
target_compile_definitions(port_test PRIVATE CAN_MOTOR_CONTROL_SERVICE=1)
target_compile_options(port_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
add_test(NAME actual_motor_port_locked_and_uart_compatibility COMMAND port_test)
